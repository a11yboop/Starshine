cmake_minimum_required(VERSION 3.24)
project(AnimeRecorder LANGUAGES CXX)
add_executable(recorder src/recorder.cpp)
target_compile_features(recorder PRIVATE cxx_std_20)
target_include_directories(recorder PRIVATE third_party)
target_link_libraries(recorder PRIVATE user32)
enable_testing()
add_test(NAME recorder_events COMMAND recorder --self-test)
foreach(test_name projection_tests placement_geometry_tests placement_binding_tests)
 add_executable(${test_name} tests/${test_name}.cpp)
 target_compile_features(${test_name} PRIVATE cxx_std_20)
 target_include_directories(${test_name} PRIVATE src third_party)
 add_test(NAME ${test_name} COMMAND ${test_name})
 set_tests_properties(${test_name} PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endforeach()
